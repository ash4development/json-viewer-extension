export function diffStats(diffs) {
  let added = 0, removed = 0, changed = 0;
  for (let i = 0; i < diffs.length; i++) {
    const d = diffs[i];

    else if (d.type === 'remove') removed++;
    else changed++;
  }
  return { added, removed, changed };
